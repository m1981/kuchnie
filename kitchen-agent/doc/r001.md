#### Step 2: Extract the Chat Store (Pattern 3)

Create a dedicated store for the chat session. This removes 60% of the code from your `+page.svelte`.

```typescript
// src/lib/stores/chat.svelte.ts
import { api } from '$lib/api';
import type { Message, PastedImage } from '$lib/types';
import type { AsyncState } from '$lib/types/states';

export function createChatStore() {
    // Private state
    let sessionId = $state(crypto.randomUUID());
    let messages = $state<Message[]>([]);
    let chatStatus = $state<AsyncState<void>>({ status: 'idle' });
    let pastedImages = $state<PastedImage[]>([]);

    return {
        // Public getters
        get sessionId() {
            return sessionId;
        },
        get messages() {
            return messages;
        },
        get status() {
            return chatStatus;
        },
        get pastedImages() {
            return pastedImages;
        },

        // Actions
        async loadSession(id: string) {
            chatStatus = { status: 'loading' };
            try {
                const data = await api.getSession(id);
                sessionId = id;
                messages = data.ui_messages || [];
                chatStatus = { status: 'success', data: undefined };
            } catch (e) {
                chatStatus = { status: 'error', message: String(e) };
            }
        },

        async sendMessage(text: string, modeId: string, contextFiles: string[]) {
            if (!text.trim() || chatStatus.status === 'loading') return;

            // Optimistic UI update
            messages.push({
                role: 'user',
                content: text,
                images: pastedImages.map((i) => i.dataUrl)
            });
            const imagesToSend = [...pastedImages];
            pastedImages = []; // clear UI immediately
            chatStatus = { status: 'loading' };

            try {
                const data = await api.chat({
                    session_id: sessionId,
                    message: text,
                    mode_id: modeId,
                    images: imagesToSend,
                    context_files: contextFiles
                });
                messages.push({ role: 'assistant', content: data.text, tools: data.tools_used });
                chatStatus = { status: 'success', data: undefined };
            } catch (e) {
                chatStatus = { status: 'error', message: String(e) };
            }
        },

        addPastedImage(img: PastedImage) {
            pastedImages.push(img);
        },
        removeImage(index: number) {
            pastedImages.splice(index, 1);
        }
    };
}

// Singleton instance for the app
export const chatStore = createChatStore();
```

#### Step 3: Extract DOM Actions (Pattern 7)

Move the `handlePaste` logic into a reusable Svelte Action.

```typescript
// src/lib/actions/pasteImage.ts
import type { Action } from 'svelte/action';

export const pasteImage: Action<HTMLElement, (img: any) => void> = (node, onImagePasted) => {
    function handlePaste(event: ClipboardEvent) {
        const items = event.clipboardData?.items;
        if (!items) return;

        for (const item of Array.from(items)) {
            if (!item.type.startsWith('image/')) continue;
            event.preventDefault();
            const file = item.getAsFile();
            if (!file) continue;

            const reader = new FileReader();
            reader.onload = (e) => {
                const dataUrl = e.target?.result as string;
                const [header, base64] = dataUrl.split(',');
                onImagePasted({ dataUrl, mimeType: header.split(':')[1].split(';')[0], base64 });
            };
            reader.readAsDataURL(file);
        }
    }

    node.addEventListener('paste', handlePaste);
    return {
        destroy() {
            node.removeEventListener('paste', handlePaste);
        }
    };
};
```

#### Step 4: Rebuild the UI Components

Now your UI components become incredibly "dumb" and clean. They just read from the store and render HTML.

**`src/lib/components/ChatComposer.svelte`**

```svelte
<script lang="ts">
    import { chatStore } from '$lib/stores/chat.svelte';
    import { pasteImage } from '$lib/actions/pasteImage';

    let { activeModeId, contextFiles } = $props();
    let currentMessage = $state('');

    function handleSend() {
        chatStore.sendMessage(currentMessage, activeModeId, contextFiles);
        currentMessage = '';
    }
</script>

<!-- Notice how clean this is! We use the action `use:pasteImage` -->
<textarea
    bind:value={currentMessage}
    use:pasteImage={chatStore.addPastedImage}
    onkeydown={(e) => e.key === 'Enter' && !e.shiftKey && (e.preventDefault(), handleSend())}
></textarea>

<button
    onclick={handleSend}
    disabled={chatStore.status.status === 'loading' || !currentMessage}
>
    Send
</button>
```
